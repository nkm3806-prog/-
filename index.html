    function goToPage(idx) {

      if (idx < 0 || idx >= PAGES.length) return;

      currentPage = idx;

      renderPage();

      closeToc();

    }

    function nextPage() { goToPage(currentPage + 1); }

    function prevPage() { goToPage(currentPage - 1); }


    function openToc() {

      const list = document.getElementById('tocList');

      list.innerHTML = PAGES.map((p, i) => `

        <button class="toc-item ${i === currentPage ? 'current' : ''}" onclick="goToPage(${i})">

          <span><strong>${p.num}.</strong> ${p.ko}</span>

          <span style="font-family:'Cormorant Garamond',serif;font-size:0.85rem;color:var(--ink-muted);">${p.en}</span>

        </button>

      `).join('');

      document.getElementById('tocModal').classList.add('open');

    }

    function closeToc() { document.getElementById('tocModal').classList.remove('open'); }

    function exportJson() {

      const blob = new Blob([JSON.stringify(appData, null, 2)], { type: 'application/json' });

      const a = document.createElement('a');

      a.href = URL.createObjectURL(blob);

      a.download = 'sparkling-frog-programbook.json';

      a.click();

    }


    function importJson(event) {

      const file = event.target.files[0];

      if (!file) return;

      const reader = new FileReader();

      reader.onload = function(e) {

        try {

          appData = JSON.parse(e.target.result);

          saveData();

          renderPage();

          closeDataModal();

          alert('데이터를 성공적으로 불러왔습니다.');

        } catch (err) {

          alert('올바른 JSON 파일이 아닙니다.');

        }

      };

      reader.readAsText(file);

    }
